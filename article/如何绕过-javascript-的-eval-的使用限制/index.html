如何绕过 javascript 的 eval 的使用限制

2020-02-15

最近读了这篇文章, 让我回忆了一下我曾经学过的 eopl3, 自己也试着写了一个超级简单的解释器, 来描述执行函数的过程是怎么样的, 看了这个过程, 也就能明白为何文章里提到的方法可以绕过 eval.

不过, 预期重新写一个 js 的解释器, 还不如直接使用 quickjs 更为方便.

文章的地址是: https://zhuanlan.zhihu.com/p/34191831

自己写的解释器如下:

    function FunctionExpression(fnName, args, body) {
        this.fnName = fnName;
        this.args = args;
        this.body = body;
    }

    function CallExpression(fnName, args) {
        this.fnName = fnName;
        this.args = args;
    }

    function VariableExpression(key) {
        this.key = key;
    }

    function NumberExpression (n) {
        this.value = Number(n);
    }

    function ConsoleExpression(expr) {
        this.expr = expr;
    }

    function findByName(env, name) {
        for (let item of env) {
            if (item.key === name) {
                return item.value;
            }
        }

        return null;
    }

    // 一个解释 js 的 js 版本的解释器
    function valueOf(env, node) {
        if (node instanceof NumberExpression) {
            return node.value;
        } else if (node instanceof VariableExpression) {
            let value = findByName(env, node.key);

            if (value) {
                return findByName(env, node.key);
            } else {
                throw new Error('变量不存在')
            }
        } else if (node instanceof ConsoleExpression) {
            let value = valueOf(env, node.expr);
            console.log('[value]', value);
        } else if (node instanceof CallExpression) {
            let func = findByName(env, node.fnName);

            if (func) {
                let args = node.args.map(function (arg, idx) {
                    let value = valueOf(env, arg);
                    return {
                        key: func.args[idx],
                        value,
                    }
                });
                valueOf(env.concat(args), func.body);
            } else {
                throw new Error('函数不存在');
            }
        } else {
            // 暂时先不管这里
        }
    }

    // 下面展示一下执行一个函数调用的过程
    // 要执行 function print(n) { console.log(n); }; print(3);
    // 为了方便, 直接手写 ast
    let fnDecl = {
        key: 'print',
        value: new FunctionExpression(
            'print',
            ['n'],
            new ConsoleExpression(new VariableExpression('n'))
        )
    };

    valueOf([fnDecl], new CallExpression('print', [new NumberExpression('3')]));
